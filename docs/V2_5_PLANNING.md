- 客户端无需额外依赖

**示例输出**:
```python
from docx import Document

def export_to_word(scan_id):
    doc = Document()
    doc.add_heading('玄武固件安全扫描报告', 0)
    
    # 添加执行摘要
    doc.add_heading('执行摘要', level=1)
    doc.add_paragraph(f"扫描时间：{scan_time}")
    doc.add_paragraph(f"发现漏洞数：{vuln_count}")
    
    # 添加漏洞表格
    table = doc.add_table(rows=1, cols=8)
    # ...填充数据
    
    doc.save(f'report_{scan_id}.docx')
```
